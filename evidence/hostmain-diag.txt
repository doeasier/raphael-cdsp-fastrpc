== host_main.c ==
/*
 * host_main.c - host (applications processor / Linux on K20 Pro) side example.
 *
 * Flow:
 *   0) Enable Unsigned PD (must be done BEFORE the first FastRPC call!) so the
 *      cDSP is allowed to load our unsigned hvx_add_skel.so.
 *   1) Prepare two int32 vectors a, b and an output vector out.
 *   2) Open a remote cDSP session via the QAIC-generated stub (hvx_add_open).
 *   3) Dispatch the vector add to the DSP (hvx_add_addVec).
 *   4) Verify the result on the host (out[i] == a[i] + b[i]).
 *   5) Close the session.
 *
 * API generated by QAIC from hvx_add.idl:
 *   - hvx_add_open(const char* uri, remote_handle64* h) -> int
 *   - hvx_add_close(remote_handle64 h) -> int
 *   - hvx_add_rpc_init(remote_handle64 h) -> int
 *   - hvx_add_addVec(remote_handle64 h, const int32* a, int aLen,
 *                    const int32* b, int bLen, int32* result, int resultLen) -> int
 *
 * Unsigned PD control (from <remote.h> / remote_default.h):
 *   int remote_session_control(uint32_t req, void* data, uint32_t datalen);
 *   struct remote_rpc_control_unsigned_module { int domain; int enable; };
 *   #define DSPRPC_CONTROL_UNSIGNED_MODULE 2
 *   #define CDSP_DOMAIN_ID 3
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* QAIC generated: prototypes + remote_handle64 + hvx_add_URI */
#include "hvx_add.h"

/* FastRPC userspace: remote_session_control(), DSPRPC_CONTROL_UNSIGNED_MODULE,
 * CDSP_DOMAIN / CDSP_DOMAIN_ID, struct remote_rpc_control_unsigned_module. */
#include <remote.h>

#ifndef CDSP_DOMAIN_ID
#define CDSP_DOMAIN_ID 3
#endif
#ifndef CDSP_DOMAIN
#define CDSP_DOMAIN "&_dom=cdsp"
#endif

/*
 * enable_unsigned_pd: allow the cDSP to load our unsigned skel .so.
 * Must run before hvx_add_open(). Non-fatal if the stack already permits it.
 */
static int enable_unsigned_pd(void) {
    struct remote_rpc_control_unsigned_module data;
    int rc;

    memset(&data, 0, sizeof(data));
    data.enable = 1;                 /* 1 = allow unsigned modules */
    data.domain = CDSP_DOMAIN_ID;    /* 3 = cDSP */

    rc = remote_session_control(DSPRPC_CONTROL_UNSIGNED_MODULE,
                                (void*)&data, (uint32_t)sizeof(data));
    if (rc != 0) {
        fprintf(stderr,
                "host: remote_session_control(UNSIGNED_MODULE) returned %d "
                "(non-fatal: may already be allowed)\n", rc);
        return 0;
    }
    printf("host: unsigned PD enabled for cDSP (domain=%d)\n", CDSP_DOMAIN_ID);
    return 0;
}

#define N        (1000)          /* vector length */
#define PRINT_N  (16)            /* print first few for eyeball check */

int main(void) {
    remote_handle64 handle = 0;
    int32_t *a = NULL, *b = NULL, *out = NULL;
    int i, rc, status = 0;
    int mismatches = 0;

    /* 0) Enable Unsigned PD - before ANY FastRPC call */
    enable_unsigned_pd();

    /* 1) Prepare data (host malloc is fine; QAIC stub copies to DSP via rpcmem) */
    a   = (int32_t*)malloc(N * sizeof(int32_t));
    b   = (int32_t*)malloc(N * sizeof(int32_t));
    out = (int32_t*)malloc(N * sizeof(int32_t));
    if (a == NULL || b == NULL || out == NULL) {
        fprintf(stderr, "host: malloc failed\n");
        status = -1;
        goto cleanup;
    }
    for (i = 0; i < N; i++) {
        a[i]   = (int32_t)(i * 2 - 500);   /* include negatives to test sign */
        b[i]   = (int32_t)(i + 1);
        out[i] = 0;
    }

    /* 2) Open remote cDSP session (CDSP selected via CDSP_DOMAIN in the URI) */
    rc = hvx_add_open(hvx_add_URI CDSP_DOMAIN, &handle);
    if (rc != 0) {
        fprintf(stderr, "host: hvx_add_open failed, rc=%d\n", rc);
        status = -2;
        goto cleanup;
    }
    printf("host: opened cDSP session, handle=%llu\n", (unsigned long long)handle);

    /* Optional one-time init (no-op here) */
    rc = hvx_add_rpc_init(handle);
    if (rc != 0) {
        fprintf(stderr, "host: rpc_init returned %d (non-fatal)\n", rc);
    }

    /* 3) Dispatch vector add to the DSP (HVX on Hexagon 690) */
    rc = hvx_add_addVec(handle, a, N, b, N, out, N);
    if (rc != 0) {
        fprintf(stderr, "host: hvx_add_addVec failed, rc=%d\n", rc);
        status = -3;
        goto close;
    }
    printf("host: hvx_add_addVec done (N=%d)\n", N);

    /* 4) Verify on host */
    for (i = 0; i < N; i++) {
        if (out[i] != a[i] + b[i]) {
            mismatches++;
            if (mismatches <= 8) {
                fprintf(stderr, "host: mismatch at %d: out=%d expected=%d\n",
                        i, out[i], a[i] + b[i]);
            }
        }
    }

    printf("host: first %d results (a + b = out):\n", PRINT_N);
    for (i = 0; i < PRINT_N && i < N; i++) {
        printf("  [%4d] %6d + %6d = %6d\n", i, a[i], b[i], out[i]);
    }

    if (mismatches == 0) {
        printf("host: PASS - all %d elements verified correct.\n", N);
    } else {
        fprintf(stderr, "host: FAIL - %d/%d mismatches.\n", mismatches, N);
        status = -4;
    }

close:
    hvx_add_close(handle);
cleanup:
    free(a);
    free(b);
    free(out);
    return status;
}
